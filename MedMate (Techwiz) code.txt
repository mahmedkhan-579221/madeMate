// ============================================================
// STOCK ALARM - IMPORTANT FIX
// ============================================================
//
// This function immediately and completely stops the stock
// alarm.
//
// This is the main fix for the problem where the buzzer
// continued after refilling the medicine stock.
// ============================================================

void stopStockAlarm() {

  stockAlarmActive = false;

  stockAlarmStart = 0;

  noTone(ALARM_PIN);
}

// ============================================================
// UPDATE STOCK ALARM STATE AFTER STOCK CHANGE
// ============================================================
//
// Whenever stock changes, this function must be called.
//
// If stock becomes healthy (> lowStockLimit), the alarm,
// alarm timer and buzzer are immediately stopped.
// ============================================================

void updateStockAlarmAfterStockChange() {

  if (
    pillStock > lowStockLimit
  ) {

    stopStockAlarm();
  }
}

// ============================================================
// ARCHIVE CURRENT DAY
// ============================================================

void archiveCurrentDay() {

  if (
    !LittleFS.exists(
      "/history.csv"
    )
  ) {

    File file =
      LittleFS.open(
        "/history.csv",
        FILE_WRITE
      );

    if (file) {

      file.println(
        "Date,Dose,Taken Time,Status"
      );

      file.close();
    }
  }

  File file =
    LittleFS.open(
      "/history.csv",
      FILE_APPEND
    );

  if (!file) {
    return;
  }

  int archiveDay =
    doses[0].day;

  int archiveMonth =
    doses[0].month;

  int archiveYear =
    doses[0].year;

  if (
    archiveDay < 0
  ) {

    archiveDay =
      currentDay;

    archiveMonth =
      currentMonth;

    archiveYear =
      currentYear;
  }

  char dateBuffer[20];

  sprintf(
    dateBuffer,
    "%04d-%02d-%02d",
    archiveYear,
    archiveMonth,
    archiveDay
  );

  for (
    int i = 0;
    i < TOTAL_DOSES;
    i++
  ) {

    String takenTime =
      "--:--:--";

    if (
      doses[i].takenHour >= 0
    ) {

      char timeBuffer[20];

      sprintf(
        timeBuffer,
        "%02d:%02d:%02d",
        doses[i].takenHour,
        doses[i].takenMinute,
        doses[i].takenSecond
      );

      takenTime =
        String(timeBuffer);
    }

    file.print(dateBuffer);
    file.print(",");
    file.print(doses[i].name);
    file.print(",");
    file.print(takenTime);
    file.print(",");
    file.println(doses[i].status);
  }

  file.close();
}

// ============================================================
// NEW DAY
// ============================================================

void checkNewDay(
  struct tm &timeinfo
) {

  int newDay =
    timeinfo.tm_mday;

  int newMonth =
    timeinfo.tm_mon + 1;

  int newYear =
    timeinfo.tm_year + 1900;

  if (
    currentDay == -1
  ) {

    currentDay =
      newDay;

    currentMonth =
      newMonth;

    currentYear =
      newYear;

    loadDoseData();

    return;
  }

  if (

    newDay != currentDay ||

    newMonth != currentMonth ||

    newYear != currentYear

  ) {

    archiveCurrentDay();

    currentDay =
      newDay;

    currentMonth =
      newMonth;

    currentYear =
      newYear;

    resetDailyData();

    saveDoseData();
  }
}

// ============================================================
// FIND NEXT INCOMPLETE DOSE
// ============================================================

int findNextIncompleteDose() {

  for (
    int i = 0;
    i < TOTAL_DOSES;
    i++
  ) {

    if (
      !doses[i].completed
    ) {

      return i;
    }
  }

  return 0;
}

// ============================================================
// CURRENT TAKEABLE DOSE
// ============================================================

int findCurrentTakeableDose(
  struct tm &timeinfo
) {

  int nowSeconds =
    getCurrentSeconds(
      timeinfo
    );

  for (
    int i = 0;
    i < TOTAL_DOSES;
    i++
  ) {

    if (
      doses[i].completed
    ) {

      continue;
    }

    int doseSeconds =
      doses[i].hour * 3600 +
      doses[i].minute * 60;

    if (
      nowSeconds >= doseSeconds
    ) {

      if (
        i == TOTAL_DOSES - 1
      ) {

        return i;
      }

      int nextDoseSeconds =
        doses[i + 1].hour * 3600 +
        doses[i + 1].minute * 60;

      if (
        nowSeconds < nextDoseSeconds
      ) {

        return i;
      }
    }
  }

  return -1;
}

// ============================================================
// MARK MISSED DOSES
// ============================================================

void markMissedDoses(
  struct tm &timeinfo
) {

  int nowSeconds =
    getCurrentSeconds(
      timeinfo
    );

  for (
    int i = 0;
    i < TOTAL_DOSES - 1;
    i++
  ) {

    if (
      doses[i].completed
    ) {

      continue;
    }

    int nextDoseSeconds =
      doses[i + 1].hour * 3600 +
      doses[i + 1].minute * 60;

    if (
      nowSeconds >= nextDoseSeconds
    ) {

      doses[i].completed =
        true;

      doses[i].status =
        "Missed";

      doses[i].takenHour =
        -1;

      doses[i].takenMinute =
        -1;

      doses[i].takenSecond =
        -1;

      doseTriggered[i] =
        false;

      if (
        activeDose == i
      ) {

        activeDose =
          -1;

        medicineAlarmMode =
          MED_ALARM_OFF;

        noTone(ALARM_PIN);
      }

      saveDoseData();
    }
  }
}

// ============================================================
// CHECK SINGLE DOSE
// ============================================================

void checkSingleDose(
  int index,
  struct tm &timeinfo
) {

  if (
    doses[index].completed
  ) {

    doseTriggered[index] =
      false;

    return;
  }

  int nowSeconds =
    getCurrentSeconds(
      timeinfo
    );

  int doseSeconds =
    doses[index].hour * 3600 +
    doses[index].minute * 60;

  int graceSeconds =
    lateWindowMinutes * 60;

  int elapsed =
    nowSeconds -
    doseSeconds;

  if (
    elapsed < 0
  ) {

    return;
  }

  if (
    elapsed <= graceSeconds
  ) {

    int currentTakeable =
      findCurrentTakeableDose(
        timeinfo
      );

    if (
      currentTakeable == index
    ) {

      activeDose =
        index;
    }

    if (
      !doseTriggered[index]
    ) {

      doseTriggered[index] =
        true;

      doseDueMillis[index] =
        millis();

      activeDose =
        index;

      medicineAlarmMode =
        MED_ALARM_INITIAL;

      medicineAlarmStart =
        millis();

      lastMedicineReminder =
        millis();

      tone(
        ALARM_PIN,
        2000
      );
    }

    return;
  }

  if (
    elapsed > graceSeconds
  ) {

    int currentTakeable =
      findCurrentTakeableDose(
        timeinfo
      );

    if (
      currentTakeable == index
    ) {

      activeDose =
        index;
    }
  }
}

// ============================================================
// CHECK ALL DOSES
// ============================================================

void checkDoses(
  struct tm &timeinfo
) {

  markMissedDoses(
    timeinfo
  );

  for (
    int i = 0;
    i < TOTAL_DOSES;
    i++
  ) {

    checkSingleDose(
      i,
      timeinfo
    );
  }

  int newActive =
    findCurrentTakeableDose(
      timeinfo
    );

  if (
    newActive >= 0
  ) {

    activeDose =
      newActive;
  }
  else {

    if (
      medicineAlarmMode ==
      MED_ALARM_OFF
    ) {

      activeDose =
        -1;
    }
  }
}

// ============================================================
// TAKE MEDICINE
// ============================================================

void takeMedicine(
  int index,
  struct tm &timeinfo
) {

  if (
    index < 0 ||
    index >= TOTAL_DOSES
  ) {

    return;
  }

  if (
    doses[index].completed
  ) {

    return;
  }

  int nowSeconds =
    getCurrentSeconds(
      timeinfo
    );

  int doseSeconds =
    doses[index].hour * 3600 +
    doses[index].minute * 60;

  int elapsed =
    nowSeconds -
    doseSeconds;

  if (
    elapsed < 0
  ) {

    lcdMedicine.clear();

    lcdMedicine.setCursor(0, 0);

    lcdMedicine.print(
      "Too Early"
    );

    lcdMedicine.setCursor(0, 1);

    lcdMedicine.print(
      doses[index].name
    );

    delay(1500);

    return;
  }

  if (
    index < TOTAL_DOSES - 1
  ) {

    int nextDoseSeconds =
      doses[index + 1].hour * 3600 +
      doses[index + 1].minute * 60;

    if (
      nowSeconds >= nextDoseSeconds
    ) {

      doses[index].completed =
        true;

      doses[index].status =
        "Missed";

      saveDoseData();

      return;
    }
  }

  doses[index].completed =
    true;

  doses[index].takenHour =
    timeinfo.tm_hour;

  doses[index].takenMinute =
    timeinfo.tm_min;

  doses[index].takenSecond =
    timeinfo.tm_sec;

  if (
    elapsed <=
    lateWindowMinutes * 60
  ) {

    doses[index].status =
      "On Time";
  }

  else {

    doses[index].status =
      "Late";
  }

  doses[index].day =
    currentDay;

  doses[index].month =
    currentMonth;

  doses[index].year =
    currentYear;

  if (
    pillStock > 0
  ) {

    pillStock--;
  }

  medicineAlarmMode =
    MED_ALARM_OFF;

  noTone(ALARM_PIN);

  activeDose =
    -1;

  doseTriggered[index] =
    false;

  saveDoseData();

  lcdMedicine.clear();

  lcdMedicine.setCursor(0, 0);

  lcdMedicine.print(
    doses[index].name
  );

  lcdMedicine.setCursor(0, 1);

  if (
    doses[index].status ==
    "On Time"
  ) {

    lcdMedicine.print(
      "On Time"
    );
  }

  else {

    lcdMedicine.print(
      "Late"
    );
  }

  delay(1500);
}

// ============================================================
// UPDATE MEDICINE ALARM
// ============================================================

void updateMedicineAlarm() {

  unsigned long now =
    millis();

  if (
    medicineAlarmMode ==
    MED_ALARM_INITIAL
  ) {

    if (
      now -
      medicineAlarmStart >=
      medicineInitialAlarmDuration
    ) {

      noTone(ALARM_PIN);

      medicineAlarmMode =
        MED_ALARM_OFF;

      lastMedicineReminder =
        now;
    }

    return;
  }

  if (
    medicineAlarmMode ==
    MED_ALARM_REMINDER
  ) {

    if (
      now -
      medicineAlarmStart >=
      medicineReminderDuration
    ) {

      noTone(ALARM_PIN);

      medicineAlarmMode =
        MED_ALARM_OFF;

      lastMedicineReminder =
        now;
    }

    return;
  }
}

// ============================================================
// MEDICINE REMINDER
// ============================================================

void checkMedicineReminder(
  struct tm &timeinfo
) {

  if (
    activeDose < 0
  ) {

    return;
  }

  if (
    activeDose >= TOTAL_DOSES
  ) {

    return;
  }

  if (
    doses[activeDose].completed
  ) {

    return;
  }

  int nowSeconds =
    getCurrentSeconds(
      timeinfo
    );

  int doseSeconds =
    doses[activeDose].hour * 3600 +
    doses[activeDose].minute * 60;

  int elapsed =
    nowSeconds -
    doseSeconds;

  if (
    elapsed >
    lateWindowMinutes * 60
  ) {

    unsigned long now =
      millis();

    if (
      medicineAlarmMode ==
      MED_ALARM_OFF
    ) {

      if (
        now -
        lastMedicineReminder >=
        medicineReminderInterval
      ) {

        medicineAlarmMode =
          MED_ALARM_REMINDER;

        medicineAlarmStart =
          now;

        tone(
          ALARM_PIN,
          2500
        );
      }
    }
  }
}

// ============================================================
// MEDICINE LEDS
// ============================================================

void updateMedicineLEDs(
  struct tm &timeinfo
) {

  int nowSeconds =
    getCurrentSeconds(
      timeinfo
    );

  for (
    int i = 0;
    i < TOTAL_DOSES;
    i++
  ) {

    int doseSeconds =
      doses[i].hour * 3600 +
      doses[i].minute * 60;

    int elapsed =
      nowSeconds -
      doseSeconds;

    bool ledOn =
      false;

    if (
      !doses[i].completed
    ) {

      if (
        elapsed >= 0 &&
        elapsed <=
        lateWindowMinutes * 60
      ) {

        ledOn =
          true;
      }
    }

    if (
      i == 0
    ) {

      digitalWrite(
        MORNING_LED_PIN,
        ledOn ? HIGH : LOW
      );
    }

    else if (
      i == 1
    ) {

      digitalWrite(
        AFTERNOON_LED_PIN,
        ledOn ? HIGH : LOW
      );
    }

    else {

      digitalWrite(
        EVENING_LED_PIN,
        ledOn ? HIGH : LOW
      );
    }
  }
}

// ============================================================
// LOW STOCK
// ============================================================
//
// FIXED:
//
// The old code changed stockAlarmActive=false when stock was
// refilled, but it did NOT always call noTone(ALARM_PIN).
//
// Therefore the buzzer could physically continue sounding.
//
// This version ALWAYS calls stopStockAlarm() when stock is
// healthy.
// ============================================================

void checkLowStock() {

  // ----------------------------------------------------------
  // HEALTHY STOCK
  // ----------------------------------------------------------

  if (
    pillStock > lowStockLimit
  ) {

    // IMPORTANT:
    // Immediately stop any existing low-stock alarm.
    stopStockAlarm();

    return;
  }

  // ----------------------------------------------------------
  // LOW STOCK
  // ----------------------------------------------------------

  if (
    pillStock <= lowStockLimit
  ) {

    if (
      !stockAlarmActive
    ) {

      unsigned long now =
        millis();

      if (
        now -
        lastStockAlarm >=
        stockAlarmInterval
      ) {

        stockAlarmActive =
          true;

        stockAlarmStart =
          now;

        lastStockAlarm =
          now;
      }
    }
  }
}

// ============================================================
// STOCK LEDS
// ============================================================

void updateStockLEDs() {

  if (
    pillStock <= lowStockLimit
  ) {

    digitalWrite(
      STOCK_LOW_LED_PIN,
      HIGH
    );

    digitalWrite(
      STOCK_OK_LED_PIN,
      LOW
    );
  }

  else {

    digitalWrite(
      STOCK_LOW_LED_PIN,
      LOW
    );

    digitalWrite(
      STOCK_OK_LED_PIN,
      HIGH
    );
  }
}

// ============================================================
// STOCK ALARM
// ============================================================

void updateStockAlarm() {

  // ----------------------------------------------------------
  // If stock is healthy, ALWAYS stop the alarm.
  // ----------------------------------------------------------

  if (
    pillStock > lowStockLimit
  ) {

    stopStockAlarm();

    return;
  }

  // ----------------------------------------------------------
  // No active stock alarm
  // ----------------------------------------------------------

  if (
    !stockAlarmActive
  ) {

    return;
  }

  // ----------------------------------------------------------
  // Medicine alarm has priority.
  //
  // Do not let stock alarm overwrite medicine alarm.
  // ----------------------------------------------------------

  if (
    medicineAlarmMode !=
    MED_ALARM_OFF
  ) {

    noTone(ALARM_PIN);

    return;
  }

  unsigned long now =
    millis();

  // ----------------------------------------------------------
  // Stock alarm is currently sounding
  // ----------------------------------------------------------

  if (
    now -
    stockAlarmStart <
    stockAlarmDuration
  ) {

    tone(
      ALARM_PIN,
      1200
    );
  }

  // ----------------------------------------------------------
  // Stock alarm finished
  // ----------------------------------------------------------

  else {

    noTone(ALARM_PIN);

    stockAlarmActive =
      false;
  }
}

// ============================================================
// BUTTON
// ============================================================

void handleButton(
  struct tm &timeinfo
) {

  bool buttonState =
    digitalRead(
      BUTTON_PIN
    );

  if (
    lastButtonState == HIGH &&
    buttonState == LOW
  ) {

    unsigned long now =
      millis();

    if (
      now -
      lastButtonPress >=
      buttonDebounce
    ) {

      lastButtonPress =
        now;

      int dose =
        findCurrentTakeableDose(
          timeinfo
        );

      if (
        dose >= 0
      ) {

        takeMedicine(
          dose,
          timeinfo
        );
      }

      else {

        lcdMedicine.clear();

        lcdMedicine.setCursor(
          0,
          0
        );

        lcdMedicine.print(
          "No Dose Due"
        );

        delay(1000);
      }
    }
  }

  lastButtonState =
    buttonState;
}

// ==========================================================
// COUNT PILLS
// ==========================================================

function countPills(){

const video =
document.getElementById(
"camera"
);

const image =
document.getElementById(
"cameraImage"
);

const canvas =
document.getElementById(
"canvas"
);

let source =
null;

let width =
0;

let height =
0;

if(
video &&
video.videoWidth > 0 &&
video.videoHeight > 0
){

source =
video;

width =
video.videoWidth;

height =
video.videoHeight;

}

else if(
capturedImage &&
capturedImage.naturalWidth > 0 &&
capturedImage.naturalHeight > 0
){

source =
capturedImage;

width =
capturedImage.naturalWidth;

height =
capturedImage.naturalHeight;

}

else{

alert(
"Please take a photo or start the camera first."
);

return;

}

const maxWidth =
800;

const scale =
Math.min(
1,
maxWidth / width
);

canvas.width =
Math.max(
1,
Math.floor(
width * scale
)
);

canvas.height =
Math.max(
1,
Math.floor(
height * scale
)
);

const ctx =
canvas.getContext(
"2d",
{
willReadFrequently:true
}
);

ctx.clearRect(
0,
0,
canvas.width,
canvas.height
);

ctx.drawImage(
source,
0,
0,
canvas.width,
canvas.height
);

const imageData =
ctx.getImageData(
0,
0,
canvas.width,
canvas.height
);

const data =
imageData.data;

const W =
canvas.width;

const H =
canvas.height;

const mask =
new Uint8Array(
W * H
);

for(
let y = 0;
y < H;
y++
){

for(
let x = 0;
x < W;
x++
){

const index =
(y * W + x) * 4;

const r =
data[index];

const g =
data[index + 1];

const b =
data[index + 2];

const brightness =
(r + g + b) / 3;

const maxRGB =
Math.max(
r,
g,
b
);

const minRGB =
Math.min(
r,
g,
b
);

const saturationDifference =
maxRGB -
minRGB;

if(
brightness > 130 &&
saturationDifference < 120
){

mask[
y * W + x
] = 1;

}

}

}

const cleaned =
new Uint8Array(
W * H
);

for(
let y = 1;
y < H - 1;
y++
){

for(
let x = 1;
x < W - 1;
x++
){

const pos =
y * W + x;

let neighbours =
0;

for(
let dy = -1;
dy <= 1;
dy++
){

for(
let dx = -1;
dx <= 1;
dx++
){

if(
dx === 0 &&
dy === 0
){

continue;

}

if(
mask[
(y + dy) * W +
(x + dx)
]
){

neighbours++;

}

}

}

if(
mask[pos] &&
neighbours >= 3
){

cleaned[pos] =
1;

}

}

}

const visited =
new Uint8Array(
W * H
);

let objects = [];

for(
let y = 0;
y < H;
y++
){

for(
let x = 0;
x < W;
x++
){

const start =
y * W + x;

if(
!cleaned[start] ||
visited[start]
){

continue;

}

const queue =
[start];

visited[start] =
1;

let area =
0;

let minX =
x;

let maxX =
x;

let minY =
y;

let maxY =
y;

while(
queue.length > 0
){

const current =
queue.pop();

const cx =
current % W;

const cy =
Math.floor(
current / W
);

area++;

if(cx < minX)
minX = cx;

if(cx > maxX)
maxX = cx;

if(cy < minY)
minY = cy;

if(cy > maxY)
maxY = cy;

const neighbours = [

[cx + 1,cy],
[cx - 1,cy],
[cx,cy + 1],
[cx,cy - 1]

];

for(
const n of neighbours
){

const nx =
n[0];

const ny =
n[1];

if(
nx < 0 ||
ny < 0 ||
nx >= W ||
ny >= H
){

continue;

}

const np =
ny * W + nx;

if(
cleaned[np] &&
!visited[np]
){

visited[np] =
1;

queue.push(np);

}

}

}

const objectWidth =
maxX - minX + 1;

const objectHeight =
maxY - minY + 1;

const objectArea =
objectWidth *
objectHeight;

const fillRatio =
area /
objectArea;

if(

area >= 80 &&

area <= 50000 &&

objectWidth >= 8 &&

objectHeight >= 8 &&

fillRatio >= 0.20

){

objects.push({

area:area,

width:objectWidth,

height:objectHeight,

fill:fillRatio

});

}

}

}

const totalPixels =
W * H;

objects =
objects.filter(
obj =>
obj.area <
totalPixels * 0.20
);

countedPills =
objects.length;

document.getElementById(
"resultNumber"
).innerText =
countedPills;

console.log(
"Detected pills:",
countedPills
);

if(
countedPills === 0
){

alert(

"No pills detected.\n\n" +

"Place the pills separately on a " +
"DARK/plain background, use good lighting, " +
"and capture another image."

);

}

}

// ============================================================
// HISTORY API
// ============================================================

void handleHistory() {

  String json =
    "[";

  if (
    !LittleFS.exists(
      "/history.csv"
    )
  ) {

    json +=
      "]";

    server.send(
      200,
      "application/json",
      json
    );

    return;
  }

  File file =
    LittleFS.open(
      "/history.csv",
      FILE_READ
    );

  bool first =
    true;

  file.readStringUntil(
    '\n'
  );

  while(
    file.available()
  ) {

    String line =
      file.readStringUntil(
        '\n'
      );

    line.trim();

    if (
      line.length() == 0
    ) {

      continue;
    }

    int p1 =
      line.indexOf(
        ','
      );

    int p2 =
      line.indexOf(
        ',',
        p1 + 1
      );

    int p3 =
      line.indexOf(
        ',',
        p2 + 1
      );

    if (
      p1 < 0 ||
      p2 < 0 ||
      p3 < 0
    ) {

      continue;
    }

    String date =
      line.substring(
        0,
        p1
      );

    String dose =
      line.substring(
        p1 + 1,
        p2
      );

    String taken =
      line.substring(
        p2 + 1,
        p3
      );

    String status =
      line.substring(
        p3 + 1
      );

    if (
      !first
    ) {

      json +=
        ",";
    }

    first =
      false;

    json +=
      "{";

    json +=
      "\"date\":\"";

    json +=
      date;

    json +=
      "\",";

    json +=
      "\"dose\":\"";

    json +=
      dose;

    json +=
      "\",";

    json +=
      "\"taken\":\"";

    json +=
      taken;

    json +=
      "\",";

    json +=
      "\"status\":\"";

    json +=
      status;

    json +=
      "\"";

    json +=
      "}";
  }

  file.close();

  json +=
    "]";

  server.send(
    200,
    "application/json",
    json
  );
}

// ============================================================
// SET DOSE TIMES
// ============================================================

void handleSetTimes() {

  if (
    !server.hasArg(
      "morning"
    ) ||

    !server.hasArg(
      "afternoon"
    ) ||

    !server.hasArg(
      "evening"
    )
  ) {

    server.send(
      400,
      "text/plain",
      "Missing dose times"
    );

    return;
  }

  String morning =
    server.arg(
      "morning"
    );

  String afternoon =
    server.arg(
      "afternoon"
    );

  String evening =
    server.arg(
      "evening"
    );

  auto parseTime =
    [](
      String value,
      int &hour,
      int &minute
    ) -> bool {

      value.trim();

      if (
        value.length() != 5 ||
        value.charAt(2) != ':'
      ) {

        return false;
      }

      String h =
        value.substring(
          0,
          2
        );

      String m =
        value.substring(
          3,
          5
        );

      for (
        int i = 0;
        i < 2;
        i++
      ) {

        if (
          !isDigit(
            h.charAt(i)
          ) ||

          !isDigit(
            m.charAt(i)
          )
        ) {

          return false;
        }
      }

      hour =
        h.toInt();

      minute =
        m.toInt();

      return

        hour >= 0 &&
        hour <= 23 &&

        minute >= 0 &&
        minute <= 59;
    };

  int mh,mm;

  int ah,am;

  int eh,em;

  if (
    !parseTime(
      morning,
      mh,
      mm
    ) ||

    !parseTime(
      afternoon,
      ah,
      am
    ) ||

    !parseTime(
      evening,
      eh,
      em
    )
  ) {

    server.send(
      400,
      "text/plain",
      "Invalid time. Use HH:MM"
    );

    return;
  }

  int morningMinutes =
    mh * 60 + mm;

  int afternoonMinutes =
    ah * 60 + am;

  int eveningMinutes =
    eh * 60 + em;

  if (

    morningMinutes >=
    afternoonMinutes ||

    afternoonMinutes >=
    eveningMinutes

  ) {

    server.send(
      400,
      "text/plain",
      "Times must be Morning < Afternoon < Evening"
    );

    return;
  }

  doses[0].hour =
    mh;

  doses[0].minute =
    mm;

  doses[1].hour =
    ah;

  doses[1].minute =
    am;

  doses[2].hour =
    eh;

  doses[2].minute =
    em;

  for (
    int i = 0;
    i < TOTAL_DOSES;
    i++
  ) {

    doseTriggered[i] =
      false;

    doseDueMillis[i] =
      0;
  }

  activeDose =
    -1;

  medicineAlarmMode =
    MED_ALARM_OFF;

  medicineAlarmStart =
    0;

  lastMedicineReminder =
    0;

  noTone(
    ALARM_PIN
  );

  saveDoseSchedule();

  server.send(
    200,
    "text/plain",
    "Dose schedule saved"
  );
}

// ============================================================
// SET STOCK
// ============================================================
//
// IMPORTANT FIX:
// After changing stock, immediately check whether the stock
// is healthy. If it is healthy, the low-stock alarm is stopped.
// ============================================================

void handleSetStock() {

  if (
    !server.hasArg(
      "value"
    )
  ) {

    server.send(
      400,
      "text/plain",
      "Missing value"
    );

    return;
  }

  int newStock =
    server.arg(
      "value"
    ).toInt();

  if (
    newStock < 0
  ) {

    newStock =
      0;
  }

  pillStock =
    newStock;

  // IMPORTANT:
  // Immediately stop alarm if refill made stock healthy.
  updateStockAlarmAfterStockChange();

  saveDoseData();

  server.send(
    200,
    "text/plain",
    "OK"
  );
}

// ============================================================
// ADD STOCK
// ============================================================
//
// IMPORTANT FIX:
// The old code only incremented stock and saved it.
// This version immediately checks and stops the alarm.
// ============================================================

void handleAddStock() {

  pillStock++;

  // Immediately stop low-stock alarm if stock is now > 5.
  updateStockAlarmAfterStockChange();

  saveDoseData();

  server.send(
    200,
    "text/plain",
    "OK"
  );
}

// ============================================================
// REMOVE STOCK
// ============================================================

void handleRemoveStock() {

  if (
    pillStock > 0
  ) {

    pillStock--;
  }

  // If stock is low, allow the low-stock system to work.
  // If stock is healthy, stop any stale alarm.
  updateStockAlarmAfterStockChange();

  saveDoseData();

  server.send(
    200,
    "text/plain",
    "OK"
  );
}

// ============================================================
// LOOP
// ============================================================

void loop() {

  server.handleClient();

  struct tm timeinfo;

  if (
    !getLocalTime(
      &timeinfo
    )
  ) {

    // Even without NTP, keep handling the web server.
    updateStockLEDs();

    // Critical stock-alarm safety:
    // if stock has been refilled, force buzzer OFF.
    if (
      pillStock > lowStockLimit
    ) {

      stopStockAlarm();
    }

    updateStockAlarm();

    delay(
      20
    );

    return;
  }

  // ==========================================================
  // DATE
  // ==========================================================

  checkNewDay(
    timeinfo
  );

  // ==========================================================
  // MEDICINE
  // ==========================================================

  checkDoses(
    timeinfo
  );

  checkMedicineReminder(
    timeinfo
  );

  updateMedicineAlarm();

  updateMedicineLEDs(
    timeinfo
  );

  handleButton(
    timeinfo
  );

  // ==========================================================
  // STOCK
  // ==========================================================

  checkLowStock();

  updateStockLEDs();

  updateStockAlarm();

  // ==========================================================
  // LCD
  // ==========================================================

  displayMedicineLCD(
    timeinfo
  );

  displayStockLCD();

  delay(
    100
  );
}
